\vspace{2ex}
\begin{enumerate}
\item Place the blinking edit cursor at the beginning of the text to be
copied by
moving it with the arrow keys
(\hspace{0.3ex}\begin{picture}(12,12)(0, 2)
\put(0,0){\framebox(12,12)[l]{$\:\scriptstyle\uparrow$}}
\end{picture}\hspace{0.3ex},
\begin{picture}(12,12)(0, 2)
\put(0,0){\framebox(12,12)[l]{$\:\scriptstyle\downarrow$}}
\end{picture}\hspace{0.3ex},
\begin{picture}(12,12)(0, 2)
\put(0,0){\framebox(12,12)[bl]{$\:\scriptstyle\leftarrow$}}
\end{picture}\hspace{0.3ex},
\hbox{\begin{picture}(12,12)(0, 2)
\put(0,0){\framebox(12,12)[bl]{$\:\scriptstyle\rightarrow$}}
\end{picture}\hspace{0.3ex})} or by clicking with the mouse
at the desired place.
\item Hold the shift key 
\framebox[11.mm][l]{\rule{0mm}{2.5mm}
\raisebox{-0.75mm}[0mm][0mm]{\includevspacePostScript[3mm]{Shift}{0.95}{0.95}}} 
or  \hspace{0.3ex}\framebox[11.mm][l]{~~~\tiny\texttt{shift}}\hspace{0.3ex}
down, without releasing it.
\item Place the blinking edit cursor at the end of the text
to be copied by
moving it with the arrow keys
(\hspace{0.3ex}\begin{picture}(12,12)(0, 2)
\put(0,0){\framebox(12,12)[l]{$\:\scriptstyle\uparrow$}}
\end{picture}\hspace{0.3ex},
\begin{picture}(12,12)(0, 2)
\put(0,0){\framebox(12,12)[l]{$\:\scriptstyle\downarrow$}}
\end{picture}\hspace{0.3ex},
\begin{picture}(12,12)(0, 2)
\put(0,0){\framebox(12,12)[bl]{$\:\scriptstyle\leftarrow$}}
\end{picture}\hspace{0.3ex},
\hbox{\begin{picture}(12,12)(0, 2)
\put(0,0){\framebox(12,12)[bl]{$\:\scriptstyle\rightarrow$}}
\end{picture}\hspace{0.3ex})} or by clicking with the mouse
at the desired place, so that the text to be copied
is shown in {\em reverse video}, that is, in white on a black background
(if by mistake you type text on the keyboard, it replaces the text
selected in reverse video; the delete key
\framebox[11.mm][l]{$\leftarrow$~~~}
removes the text selected in reverse video; in both cases,
choose the {\idt Undo} item
of the {\idt Edit} menu to restore the original).
%
\item Release the shift key
\framebox[11.mm][l]{\rule{0mm}{2.5mm}
\raisebox{-0.75mm}[0mm][0mm]{\includevspacePostScript[3mm]{Shift}{0.95}{0.95}}}$\,$ or  \hspace{0.3ex}\framebox[11.mm][l]{~~~\tiny\texttt{shift}}\hspace{0.3ex}.
\item Select the {\idt Copy} item of the {\idt Edit} menu
(or, equivalently, hold
the command key
\framebox[9.mm][l]{\rule{0mm}{2.5mm}
\raisebox{-0.5mm}[0mm][0mm]{\includevspacePostScript[3mm]{Cmd}{0.95}{0.95}}}
down and press the key 
\begin{picture}(12,12)(0, 2) %touche d'echappement
\put(0,0){\framebox(12,12)[lb]{}}
\put(1.5,2){\makebox(0,0)[lb]{\tiny C}}
\end{picture}\hspace{0.3ex}).
%
\item Place the blinking edit cursor at the place where the text
is to be copied by
moving it with the arrow keys
(\hspace{0.3ex}\begin{picture}(12,12)(0, 2)
\put(0,0){\framebox(12,12)[l]{$\:\scriptstyle\uparrow$}}
\end{picture}\hspace{0.3ex},
\begin{picture}(12,12)(0, 2)
\put(0,0){\framebox(12,12)[l]{$\:\scriptstyle\downarrow$}}
\end{picture}\hspace{0.3ex},
\hbox{\begin{picture}(12,12)(0, 2)
\put(0,0){\framebox(12,12)[bl]{$\:\scriptstyle\leftarrow$}}
\end{picture}\hspace{0.3ex},}
\begin{picture}(12,12)(0, 2)
\put(0,0){\framebox(12,12)[bl]{$\:\scriptstyle\rightarrow$}}
\end{picture}\hspace{0.3ex}) or by clicking at the desired
place with the mouse.
\item Select the {\idt Paste} item of the {\idt Edit} menu
(or, equivalently, hold
the command key 
\framebox[9.mm][l]{\rule{0mm}{2.5mm}
\raisebox{-0.5mm}[0mm][0mm]{\includevspacePostScript[3mm]{Cmd}{0.95}{0.95}}}
down and press the key 
\begin{picture}(12,12)(0, 2) %touche d'echappement
\put(0,0){\framebox(12,12)[lb]{}}
\put(1.5,2){\makebox(0,0)[lb]{\tiny V}}
\end{picture}\hspace{0.3ex}).
\end{enumerate}
The copying operation ({\idt Paste}) at any place in the
program can be repeated as many
times as necessary.
